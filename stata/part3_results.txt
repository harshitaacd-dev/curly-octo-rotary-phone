. program define platgrowth
  1.     args plat res
  2.     use "stata/platform_month_panel.dta", clear
  3.     drop if suspect_duplicate == 1
  4.     keep if platform == "`plat'" & month >= "2022-01" & month <= "2024-12"
  5.     generate target = inlist(category_group, "sexual", "synthetic")
  6.     collapse (sum) count, by(month target)
  7.     generate mdate = monthly(month, "YM")
  8.     format mdate %tm
  9.     generate double t = (mdate - tm(2022m1)) / 12
 10.     generate moy = month(dofm(mdate))
 11.     generate double x = target * t
 12.     * main: sexual-content complaints relative to the platform's other complaints
.     quietly ppmlhdfe count x, absorb(target mdate) vce(robust)
 13.     local rt = _b[x]
 14.     local rst = _se[x]
 15.     local pr = 2 * normal(-abs((`rt' - rb) / sqrt(`rst'^2 + rsb^2)))
 16.     * secondary: raw growth of sexual-content complaints, Newey-West SE (6 lags)
.     keep if target == 1
 17.     tsset mdate
 18.     quietly glm count t i.moy, family(poisson) vce(hac nwest 6)
 19.     local g = _b[t]
 20.     local sg = _se[t]
 21.     local praw = 2 * normal(-abs((`g' - nb) / sqrt(`sg'^2 + nsb^2)))
 22.     post `res' ("`plat'") (exp(`rt')) (exp(`rt' - 1.96 * `rst')) (exp(`rt' + 1.96 * `rst')) (`pr') (exp(`g')) (`praw')
 23.     display "`plat'" _col(13) "share-adjusted " %6.3f exp(`rt') " [" %5.2f exp(`rt' - 1.96 * `rst') "-" ///
>         %5.2f exp(`rt' + 1.96 * `rst') "]  p vs police " %6.4f `pr' ///
>         "  |  raw " %7.3f exp(`g') "  p vs police " %6.4f `praw'
 24. end
